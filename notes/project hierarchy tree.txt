└── class DatabaseManager
    │
    ├── __init__()
    │   ├── get MongoDB connection string
    │   ├── MongoClient(...)
    │   └── init_database()
    │
    ├── init_database()
    │   ├── create/select database
    │   ├── cars collection
    │   ├── maintenance_records collection
    │   ├── create unique index
    │   │   └── cars.car_plate
    │   └── create index
    │       └── maintenance_records.car_id
    │
    │
    ├── CAR FUNCTIONS
    │
    ├── create_car(...)
    │   ├── build car_data dictionary
    │   ├── cars.insert_one(car_data)
    │   └── return inserted_id
    │
    ├── get_all_cars()
    │   ├── cars.find()
    │   ├── loop through cars
    │   ├── convert _id to string
    │   └── return cars
    │
    ├── get_car(car_id)
    │   ├── ObjectId(car_id)
    │   ├── cars.find_one(...)
    │   ├── if car found
    │   │   └── convert _id to string
    │   └── return car
    │
    ├── update_car(car_id, ...)
    │   ├── build update_data
    │   ├── cars.update_one(...)
    │   └── return result.modified_count > 0
    │
    ├── delete_car(car_id)
    │   ├── delete maintenance records for car
    │   ├── cars.delete_one(...)
    │   └── return delete result
    │
    │
    ├── MAINTENANCE RECORD FUNCTIONS
    │
    ├── create_maintenance_record(...)
    │   ├── build maintenance_data dictionary
    │   ├── maintenance_records.insert_one(...)
    │   └── return inserted_id
    │
    ├── get_car_maintenance_records(car_id)
    │   ├── maintenance_records.find(...)
    │   ├── loop through records
    │   ├── convert _id to string
    │   └── return records
    │
    ├── get_maintenance_record(record_id)
    │   ├── ObjectId(record_id)
    │   ├── maintenance_records.find_one(...)
    │   ├── if record found
    │   │   └── convert _id to string
    │   └── return record
    │
    ├── update_maintenance_record(record_id, ...)
    │   ├── build update_data
    │   ├── maintenance_records.update_one(...)
    │   └── return result.modified_count > 0
    │
    ├── delete_maintenance_record(record_id)
    │   ├── maintenance_records.delete_one(...)
    │   └── return delete result
    │
    └── close_connection()
        └── close MongoDB client

│   ├── car_id

│   ├── cost
│   └── notes

│   └── maintenance update fields
│

│   ├── record_id

│   └── created_at
│
├── DATABASE CONNECTION
│   └── try
│       ├── db = DatabaseManager()

│   └── root()

│
├── CAR API
│
├── create_car(...)
│   └── POST /cars/
│       ├── db.create_car(...)
│       ├── if successful
│       │   └── return result
│       └── else
│           └── raise HTTPException
│
├── get_all_cars()
│   └── GET /cars/
│       ├── db.get_all_cars()
│       ├── loop through cars
│       └── return CarResponse
│

│   └── GET /cars/{car_id}
│       ├── validate ID
│       ├── db.get_car(car_id)
│       ├── if car found
│       │   └── return CarResponse
│       └── else
│           └── raise HTTPException
│
├── update_car(car_id, car_update)
│   └── PUT /cars/{car_id}
│       ├── validate ID
│       ├── db.update_car(...)
│       ├── if successful
│       │   └── return success
│       └── else
│           └── raise HTTPException
│
├── delete_car(car_id)
│   └── DELETE /cars/{car_id}
│       ├── validate ID
│       ├── db.delete_car(car_id)
│       ├── if successful
│       │   └── return success
│       └── else
│           └── raise HTTPException
│
├── MAINTENANCE RECORD API
│
├── create_maintenance_record(...)
│   └── POST /maintenance/
│       ├── validate car_id
│       ├── db.create_maintenance_record(...)
│       ├── if successful
│       │   └── return record_id
│       └── else
│           └── raise HTTPException
│
├── get_car_maintenance_records(car_id)
│   └── GET /maintenance/car/{car_id}
│       ├── validate car_id
│       ├── db.get_car_maintenance_records(car_id)
│       └── return records
│
├── get_maintenance_record(record_id)
│   └── GET /maintenance/{record_id}
│       ├── validate record_id
│       ├── db.get_maintenance_record(record_id)
│       └── return record
│
├── update_maintenance_record(record_id, ...)
│   └── PUT /maintenance/{record_id}
│       ├── validate record_id
│       ├── db.update_maintenance_record(...)
│       ├── if successful
│       │   └── return success
│       └── else
│           └── raise HTTPException
│
└── delete_maintenance_record(record_id)

        ├── validate record_id
        ├── db.delete_maintenance_record(record_id)
        ├── if successful
        │   └── return success
        └── else
            └── raise HTTPException

IMPORTANT: HOW THE 3 FILES CONNECT
==================================================

USER
│
└── Streamlit
    │
    └── streamlit_app.py
        │
        └── requests
            │
            ▼
        FastAPI
        │
        └── main.py
            │
            └── DatabaseManager
                │
                ▼
            database.py
                │
                ▼
            MongoDB Atlas


==================================================
EXAMPLE: CREATE MAINTENANCE RECORD FLOW
==================================================

User fills form
│
└── streamlit_app.py
    │
    └── create_maintenance_record(...)
        │
        └── POST request
            │
            ▼
        main.py
        │
        └── POST /maintenance/
            │
            └── MaintenanceRecordCreate
                │
                └── db.create_maintenance_record(...)
                    │
                    ▼
                database.py
                    │
                    └── maintenance_records.insert_one(...)
                        │
                        ▼
                    MongoDB Atlas


==================================================
EXAMPLE: VIEW MAINTENANCE RECORD FLOW
==================================================

User selects car
│
└── streamlit_app.py
    │
    ├── get car_id from car_options
    │
    └── get_car_maintenance_records(car_id)
        │
        └── GET request
            │
            ▼
        main.py
        │
        └── GET /maintenance/car/{car_id}
            │
            └── db.get_car_maintenance_records(car_id)
                │
                ▼
            database.py
                │
                └── maintenance_records.find(...)
                    │
                    ▼
                MongoDB Atlas
                    │
                    ▼
                records returned
                    │
                    ▼
                main.py
                    │
                    ▼
                streamlit_app.py
                    │
                    └── display records

        ↓

        ↓
